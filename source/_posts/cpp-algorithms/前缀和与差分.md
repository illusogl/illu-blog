前缀和把区间和变成两个前缀相减；差分把区间加变成两个端点的修改。一维和二维的公式只差容斥，写的时候注意下标从 1 开始、差分数组多开一位。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

// 1D 前缀和：pre[i] = a[1] + ... + a[i]
vector<ll> buildPrefix(const vector<ll>& a)
{
    int n = (int)a.size() - 1;
    vector<ll> pre(n + 1, 0);
    for (int i = 1; i <= n; i++)
        pre[i] = pre[i - 1] + a[i];
    return pre;
}

// 1D 区间和：a[l] + ... + a[r]
ll rangeSum(const vector<ll>& pre, int l, int r)
{
    return pre[r] - pre[l - 1];
}

// 1D 差分：diff[i] = a[i] - a[i - 1]
// 区间加：diff[l] += v; diff[r + 1] -= v;
// 还原：for (int i = 1; i <= n; i++) diff[i] += diff[i - 1];
vector<ll> buildDiff(const vector<ll>& a)
{
    int n = (int)a.size() - 1;
    vector<ll> diff(n + 2, 0);
    for (int i = 1; i <= n; i++)
        diff[i] = a[i] - a[i - 1];
    return diff;
}

// 2D 前缀和：pre[i][j] = 左上角 (1,1) 到 (i,j) 的矩形和
vector<vector<ll>> buildPrefix2D(const vector<vector<ll>>& a)
{
    int n = (int)a.size() - 1;
    int m = (int)a[1].size() - 1;
    vector<vector<ll>> pre(n + 1, vector<ll>(m + 1, 0));
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            pre[i][j] = pre[i - 1][j] + pre[i][j - 1] - pre[i - 1][j - 1] + a[i][j];
    return pre;
}

// 2D 矩形和：(x1,y1) 到 (x2,y2)
ll rectSum(const vector<vector<ll>>& pre, int x1, int y1, int x2, int y2)
{
    return pre[x2][y2] - pre[x1 - 1][y2] - pre[x2][y1 - 1] + pre[x1 - 1][y1 - 1];
}

// 2D 差分：给矩形 (x1,y1) 到 (x2,y2) 加 v
// diff 要开 (n + 2) * (m + 2)，最后做一次 2D 前缀和还原
void addRect(vector<vector<ll>>& diff, int x1, int y1, int x2, int y2, ll v)
{
    diff[x1][y1] += v;
    diff[x1][y2 + 1] -= v;
    diff[x2 + 1][y1] -= v;
    diff[x2 + 1][y2 + 1] += v;
}

void solve()
{
    int n, q;
    cin >> n >> q;

    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<ll> pre = buildPrefix(a);
    while (q--)
    {
        int l, r;
        cin >> l >> r;
        cout << rangeSum(pre, l, r) << '\n';
    }
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

区间和、矩形和查询 O(1)；差分修改 O(1)，还原 O(n) 或 O(nm)。适合静态区间查询、区间加、二维矩阵加。
