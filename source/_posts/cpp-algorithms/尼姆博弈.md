尼姆博弈里，每堆石子数异或起来，非零则先手必胜，为零则先手必败。prefXor(n) 返回 1 到 n 的异或，遇到“每堆是一个区间 [l, r]”的变体时，用 prefXor(r) ^ prefXor(l - 1) 求区间异或。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

// 1 到 n 的异或；区间 [l, r] 的异或 = prefXor(r) ^ prefXor(l - 1)
// 1..n 的异或按 n % 4 循环
ll prefXor(ll n)
{
    switch (n % 4)
    {
        case 0: return n;
        case 1: return 1;
        case 2: return n + 1;
        default: return 0;
    }
}

void solve()
{
    int n;
    cin >> n;

    ll xr = 0;
    for (int i = 0; i < n; i++)
    {
        ll a;
        cin >> a;
        xr ^= a; // 所有堆的异或和
    }

    // 异或和非零先手必胜
    cout << (xr ? "first" : "second") << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

判断尼姆游戏及区间尼姆的胜负。标准尼姆每组数据 O(n)，空间 O(1)；prefXor 是 O(1)。
