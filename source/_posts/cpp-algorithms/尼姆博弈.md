尼姆博弈里，每堆石子数异或起来，非零则先手必胜，为零则先手必败。这份代码处理的是每堆为一个区间 [x, x+m-1] 的情况，需要快速求一段连续整数的异或。1 到 n 的异或按 n mod 4 有固定结果，区间异或就是两个前缀异或再异或。

<!-- more -->

```cpp
ll F1n(ll n)
{
    //对1~n求异或的解很特殊，通过观察可得
    switch (n % 4)
    {
    case 0:
        return n;
    case 1:
        return 1;
    case 2:
        return n + 1;
    case 3:
        return 0;
    }
}
void solve()
{
    int n;
    cin >> n;
    vector<PLL> arr(n);
    for (int i = 0; i < n; i++)
    {
        ll x, m;
        cin >> x >> m;
        arr[i] = { m,x };
    }
    ll ans = 0;
    for (int i = 0; i < n; i++)
    {
        ans ^= F1n(arr[i].second + arr[i].first - 1) ^ F1n(arr[i].second - 1);
    }
    if (ans)
    {
        cout << "tolik" << "\n";
    }
    else
    {
        cout << "bolik" << "\n";
    }
}
```

判断尼姆游戏及区间尼姆的胜负。每组数据 O(n)，空间 O(n)。
