从 2 开始往后扫，遇到一个没被标记的数就认为它是素数，然后把它的倍数全部标记成合数。倍数从 i*i 开始，因为更小的倍数已经被更小的素因子筛过了。1 不是素数，初始化时要单独把 isPrime[1] 置为 false。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

vector<int> primes;
vector<bool> isPrime;

// 筛出 2..n 的所有素数
void sieve(int n)
{
    isPrime.assign(n + 1, true);
    if (n >= 0) isPrime[0] = false; // 0 不是素数
    if (n >= 1) isPrime[1] = false; // 1 不是素数
    primes.clear();

    for (int i = 2; i <= n; i++)
    {
        if (!isPrime[i]) continue; // 已经被更小的素数筛掉
        primes.push_back(i);
        if (1LL * i * i > n) continue; // i*i 超过 n，后面没有倍数可筛
        // 从 i*i 开始，更小的倍数已经被更小的素因子筛过
        for (ll j = 1LL * i * i; j <= n; j += i)
            isPrime[j] = false;
    }
}

void solve()
{
    int n;
    cin >> n;
    sieve(n);

    cout << primes.size() << '\n';
    for (int p : primes) cout << p << ' ';
    cout << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

预处理素数表、判断素数、分解质因数。时间复杂度 O(n log log n)，空间 O(n)。sieve(n) 的上界按题目给，筛完 primes 里就是素数表。
