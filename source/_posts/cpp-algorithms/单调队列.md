单调队列维护一个窗口内的最值。队列存下标，值从队头到队尾单调；新元素进来时，把队尾不可能成为答案的弹掉，队头过期也弹掉。队头就是当前窗口的最值。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

// 滑动窗口最大值：窗口长度 k，返回每个窗口的最大值
vector<int> slidingMax(const vector<int>& a, int k)
{
    int n = (int)a.size() - 1;
    vector<int> res;
    deque<int> dq; // 存下标，a[dq] 单调递减

    for (int i = 1; i <= n; i++)
    {
        // 队尾比当前值小的，不可能再成为最大值
        while (!dq.empty() && a[dq.back()] <= a[i]) dq.pop_back();
        dq.push_back(i);

        // 队头已经滑出窗口
        if (dq.front() <= i - k) dq.pop_front();

        // 窗口填满后记录答案
        if (i >= k) res.push_back(a[dq.front()]);
    }
    return res;
}

// 求最小值：把上面的 <= 改成 >= 即可

void solve()
{
    int n, k;
    cin >> n >> k;

    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<int> ans = slidingMax(a, k);
    for (int x : ans) cout << x << ' ';
    cout << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

滑动窗口最值、定长区间 DP 优化。每个元素最多进队出队一次，时间复杂度 O(n)，空间 O(k)。DP 优化时把队列里的值换成 dp 值，转移形如 `dp[i] = max(dp[j]) + w[i]`。
