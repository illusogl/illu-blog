回溯是带撤销的 DFS。递归到某一层时，先判断是否满足终止条件，满足就记录答案并返回；否则枚举这一层能做的选择，做选择、递归、再撤销选择，回到进入这一层时的状态。下面以全排列为例。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int n;
vector<int> path;
vector<bool> used;

void dfs()
{
    if ((int)path.size() == n) // 终止条件：已经选满 n 个数
    {
        for (int x : path) cout << x << ' ';
        cout << '\n';
        return;
    }

    for (int x = 1; x <= n; x++)
    {
        if (used[x]) continue; // 这个数已经用过
        // 做选择
        used[x] = true;
        path.push_back(x);
        dfs(); // 递归下一层
        // 撤销选择，回到上一层
        path.pop_back();
        used[x] = false;
    }
}

void solve()
{
    cin >> n;
    used.assign(n + 1, false);
    dfs();
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

用来枚举排列、组合、子集，也用于棋盘、分割、路径这类搜索题。复杂度取决于搜索树的大小，通常是指数级；剪枝和可行性判断能减少很多分支。递归栈和标记数组是 O(n)。
