线段树把区间二分，每个节点存一段区间的和，另外记一个懒标记。区间加时，如果当前节点被完全覆盖，就更新它的和并打上标记，不再往下走；只覆盖一部分时，先把标记下推给左右儿子，再递归处理。区间查询同理，完全覆盖就直接返回节点里的和。

```cpp
#include <bits/stdc++.h>
using namespace std;

/**
 * 线段树（区间和 + 懒标记）
 *
 * 数据结构：vector<pair<int,int>> tree
 *   - tree[node].first  = 该节点区间的和
 *   - tree[node].second = 该节点的懒标记（待下推的累加值）
 *
 * 功能：区间加、区间求和
 * 时间复杂度：O(log n) / 次操作
 * 空间复杂度：O(4n)
 */

class SegTree {
private:
    int n;                              // 原数组长度
    vector<pair<int, int>> tree;        // tree[node] = {区间和, 懒标记}
    //数字太大，需要动态创建时可以用unordered_map
    // 下推懒标记
    // 将 node 的懒标记传给左右子节点，并清空当前节点标记
    void pushdown(int node, int l, int r) {
        if (tree[node].second == 0) return;     // 无标记，直接返回

        int mid = (l + r)/2;
        int lazy = tree[node].second;           // 当前节点的累加值

        // 左子节点  [l, mid]
        tree[node*2].first += lazy * (mid - l + 1);
        tree[node*2].second += lazy;

        // 右子节点  [mid + 1, r]
        tree[node*2 + 1].first += lazy * (r - mid);
        tree[node*2 + 1].second += lazy;

        // 清空当前节点标记（标记已下推）
        tree[node].second = 0;
    }
    // 建树  O(n)
    // 递归构建线段树，叶子节点赋值为 arr[l]
    void build(int node, int l, int r, const vector<int>& arr) {
        if (l == r) {                           // 叶子节点
            tree[node].first = arr[l];
            tree[node].second = 0;
            return;
        }
        int mid = (l + r)/2;
        build(node*2, l, mid, arr);          // 构建左子树
        build(node*2 + 1, mid + 1, r, arr);  // 构建右子树
        tree[node].first = tree[node*2].first + tree[node*2 + 1].first;   // 合并
    }
    // 区间加  [ql, qr] 全部加上 val     O(log n)
    void range_add(int node, int l, int r, int ql, int qr, int val) {
        if (ql <= l && r <= qr) {               // 当前区间完全被覆盖
            tree[node].first += val * (r - l + 1);
            tree[node].second += val;           // 打上懒标记，不继续下传
            return;
        }
        pushdown(node, l, r);                   // 部分覆盖，先下推标记再递归
        int mid = (l + r)/2;
        if (ql <= mid) range_add(node*2, l, mid, ql, qr, val);
        if (qr > mid)  range_add(node*2 + 1, mid + 1, r, ql, qr, val);
        tree[node].first = tree[node*2].first + tree[node*2 + 1].first;   // 合并
    }
    // 区间求和  [ql, qr]     O(log n)
    int range_query(int node, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) {               // 当前区间完全被覆盖
            return tree[node].first;
        }
        pushdown(node, l, r);                   // 部分覆盖，下推标记再查询
        int mid = (l + r)/ 2;
        int res = 0;
        if (ql <= mid) res += range_query(node*2, l, mid, ql, qr);
        if (qr > mid)  res += range_query(node*2 + 1, mid + 1, r, ql, qr);
        return res;
    }

public:
    // 构造函数：用原数组 arr 初始化线段树
    SegTree(const vector<int>& arr) {
        n = arr.size();
        tree.resize(4 * n, {0, 0});     // 动态分配 4n 空间，自动管理内存
        build(1, 0, n - 1, arr);
    }
    // 对外接口：区间加
    // 将 [l, r] 每个元素加上 val
    void add(int l, int r, int val) {
        range_add(1, 0, n - 1, l, r, val);
    }
    // 对外接口：区间求和
    // 返回 [l, r] 的元素和
    int sum(int l, int r) {
        return range_query(1, 0, n - 1, l, r);
    }
};
// 使用示例
void solve() {
    int n, m;               // n: 数组大小, m: 操作次数
    cin >> n >> m;

    vector<int> arr(n);
    for (int i = 0; i < n; i++) cin >> arr[i];

    SegTree seg(arr);       // 创建线段树

    while (m--) {
        int op, l, r, val;
        cin >> op;

        if (op == 1) {                      // 操作1: 区间加
            cin >> l >> r >> val;
            l--, r--;                       // 转为 0-index（视题目要求可选）
            seg.add(l, r, val);
        } else {                            // 操作2: 区间求和
            cin >> l >> r;
            l--, r--;                       // 转为 0-index（视题目要求可选）
            cout << seg.sum(l, r) << '\n';
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    solve();
    return 0;
}
```

处理区间加、区间求和，改一改合并方式也能做区间最值、区间赋值。每次操作 O(log n)，空间 O(4n)。
