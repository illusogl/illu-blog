线段树把区间二分，每个节点存一段区间的和，另外记一个懒标记。区间加时，如果当前节点被完全覆盖，就更新它的和并打上标记，不再往下走；只覆盖一部分时，先把标记下推给左右儿子，再递归处理。区间查询同理，完全覆盖就直接返回节点里的和。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct SegTree
{
    int n;
    vector<ll> sum, lazy;

    SegTree(const vector<ll>& a)
    {
        n = (int)a.size() - 1;
        sum.assign(4 * n + 5, 0);
        lazy.assign(4 * n + 5, 0);
        build(1, 1, n, a);
    }

    // 建树：叶子放 a[l]，父亲合并左右儿子
    void build(int u, int l, int r, const vector<ll>& a)
    {
        if (l == r) // 叶子节点
        {
            sum[u] = a[l];
            return;
        }
        int mid = (l + r) / 2;
        build(u * 2, l, mid, a);
        build(u * 2 + 1, mid + 1, r, a);
        sum[u] = sum[u * 2] + sum[u * 2 + 1]; // 合并左右儿子
    }

    // 把当前节点的懒标记传给左右儿子
    void pushdown(int u, int l, int r)
    {
        if (lazy[u] == 0) return;
        int mid = (l + r) / 2;
        ll v = lazy[u];
        // 左儿子区间长度是 mid - l + 1
        sum[u * 2] += v * (mid - l + 1);
        lazy[u * 2] += v;
        sum[u * 2 + 1] += v * (r - mid);
        lazy[u * 2 + 1] += v;
        lazy[u] = 0;
    }

    // 区间加：完全覆盖就打标记，否则下推后递归
    void add(int u, int l, int r, int ql, int qr, ll v)
    {
        if (ql <= l && r <= qr) // 完全覆盖，直接打标记
        {
            sum[u] += v * (r - l + 1);
            lazy[u] += v;
            return;
        }
        pushdown(u, l, r); // 部分覆盖，先下推
        int mid = (l + r) / 2;
        if (ql <= mid) add(u * 2, l, mid, ql, qr, v);
        if (qr > mid) add(u * 2 + 1, mid + 1, r, ql, qr, v);
        sum[u] = sum[u * 2] + sum[u * 2 + 1];
    }

    // 区间查询：完全覆盖直接返回，否则下推后递归
    ll query(int u, int l, int r, int ql, int qr)
    {
        if (ql <= l && r <= qr) return sum[u]; // 完全覆盖
        pushdown(u, l, r);
        int mid = (l + r) / 2;
        ll res = 0;
        if (ql <= mid) res += query(u * 2, l, mid, ql, qr);
        if (qr > mid) res += query(u * 2 + 1, mid + 1, r, ql, qr);
        return res;
    }

    void add(int l, int r, ll v)
    {
        add(1, 1, n, l, r, v);
    }

    ll query(int l, int r)
    {
        return query(1, 1, n, l, r);
    }
};

void solve()
{
    int n, q;
    cin >> n >> q;

    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    SegTree seg(a);
    while (q--)
    {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1)
        {
            ll v;
            cin >> v;
            seg.add(l, r, v);
        }
        else
        {
            cout << seg.query(l, r) << '\n';
        }
    }
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

处理区间加、区间求和，改一改合并方式也能做区间最值、区间赋值。每次操作 O(log n)，空间 O(4n)。
