单调栈用来找每个元素左边或右边第一个比它大（或小）的位置。以“左边第一个严格更大值”为例：维护一个从栈底到栈顶递减的下标栈，新元素进来时，把栈顶小于等于它的都弹掉，剩下的栈顶就是答案。求更小值时把比较方向反过来。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

// 左边第一个严格大于 a[i] 的位置，0 表示没有
vector<int> previousGreater(const vector<ll>& a)
{
    int n = (int)a.size() - 1;
    vector<int> res(n + 1, 0);
    stack<int> st;

    for (int i = 1; i <= n; i++)
    {
        // 弹掉不可能成为答案的元素
        while (!st.empty() && a[st.top()] <= a[i]) st.pop();
        res[i] = st.empty() ? 0 : st.top(); // 栈顶就是左边第一个更大值
        st.push(i);
    }
    return res;
}

// 右边第一个大于等于 a[i] 的位置，n+1 表示没有
vector<int> nextGreater(const vector<ll>& a)
{
    int n = (int)a.size() - 1;
    vector<int> res(n + 1, n + 1);
    stack<int> st;

    for (int i = n; i >= 1; i--)
    {
        // 从右往左，弹掉更小的
        while (!st.empty() && a[st.top()] < a[i]) st.pop();
        res[i] = st.empty() ? n + 1 : st.top(); // 右边第一个大于等于 a[i]
        st.push(i);
    }
    return res;
}

// 求更小值：把上面的 <= 改成 >=，< 改成 >
// 子数组最值贡献：a[i] * (i - left[i]) * (right[i] - i)

void solve()
{
    int n;
    cin >> n;

    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<int> l = previousGreater(a);
    vector<int> r = nextGreater(a);
    for (int i = 1; i <= n; i++)
        cout << l[i] << ' ' << r[i] << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

求每个元素作为最值能扩展的区间、所有子数组最大值减最小值之和、柱状图最大矩形。每个元素最多进栈出栈一次，时间复杂度 O(n)，空间 O(n)。
