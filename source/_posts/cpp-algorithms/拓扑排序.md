拓扑排序每次取出入度为 0 的点，删掉它的出边，再把新入度为 0 的点入队。队列空时还有点没输出，说明有环。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve()
{
    int n, m;
    cin >> n >> m;

    vector<vector<int>> g(n + 1);
    vector<int> indeg(n + 1, 0);
    for (int i = 0; i < m; i++)
    {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        indeg[v]++;
    }

    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (indeg[i] == 0) q.push(i); // 入度为 0 的点先入队

    vector<int> order;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        order.push_back(u);

        for (int v : g[u])
        {
            if (--indeg[v] == 0) q.push(v); // 删掉 u 后，v 的入度减一
        }
    }

    if ((int)order.size() != n)
    {
        cout << "cycle\n"; // 还有没输出的点，说明有环
        return;
    }

    for (int x : order) cout << x << ' ';
    cout << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

DAG 上 DP、依赖关系、判环、字典序最小拓扑序（把队列换成优先队列）。时间复杂度 O(n+m)，空间 O(n+m)。
