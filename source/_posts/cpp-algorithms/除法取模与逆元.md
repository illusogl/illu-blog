模意义下不能直接做除法。(a / b) mod p 在 p 为质数时可以改成 a * b^(p-2) mod p，这里用了费马小定理。qpow 是带模的快速幂，指数按二进制拆开，每一步都取模，避免溢出。前提是 p 为质数，且 b 不是 p 的倍数。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const ll MOD = 998244353;

// 带模快速幂
ll qpow(ll a, ll b)
{
    ll res = 1 % MOD;
    while (b > 0)
    {
        if (b & 1) res = (__int128)res * a % MOD; // 防 long long 乘法溢出
        a = (__int128)a * a % MOD; // 底数平方
        b >>= 1;
    }
    return res;
}

ll inv(ll x)
{
    // 费马小定理：x^(p-2) 是 x 的逆元
    return qpow(x, MOD - 2);
}

void solve()
{
    ll a, b;
    cin >> a >> b;
    cout << a * inv(b) % MOD << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

计算组合数、概率期望、分数取模。快速幂 O(log p)，空间 O(1)。
