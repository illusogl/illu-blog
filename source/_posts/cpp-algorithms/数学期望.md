期望等于每个可能取值乘上它的概率再求和。m 面骰子掷 n 次，最大值不超过 i 的概率是 (i/m)^n。把 i 和 i-1 的概率相减，就得到最大值恰好为 i 的概率，再乘 i 累加即可。

<!-- more -->

```cpp
//https://codeforces.com/problemset/problem/453/A
//期望值的标准求法：
//Σ可能值*概率
#include <bits/stdc++.h>
using namespace std;
#define ll long long
void solve()
{
    ll n,m;
    //m面骰子，骰n次
    cin >> m >> n;
    double ans = 0;
    vector<double> index(m + 1);
    for(ll i = 1; i <= m; i++)
    {
        index[i] = pow(i*1.0/m,n);
        ans += i * (index[i] - index[i - 1]);
    }
    cout << fixed << setprecision(9) << ans << "\n";
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
        solve();
    return 0;
}
```

计算离散随机变量的期望，也常用于期望 DP 和概率题。时间复杂度 O(m log n)，空间 O(m)。
