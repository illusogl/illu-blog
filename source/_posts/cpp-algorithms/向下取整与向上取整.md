C++ 的整数除法向零取整，遇到负数就和数学上的向下取整不一致。修正方法是看余数是否为零，以及 a 和 b 是否异号：向下取整在余数不为零且异号时减一，向上取整在余数不为零且同号时加一。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

ll floorDiv(ll a, ll b)
{
    // 异号且除不尽时，向零取整比向下取整大 1
    return a / b - (a % b != 0 && (a ^ b) < 0);
}

ll ceilDiv(ll a, ll b)
{
    // 同号且除不尽时，向上取整要比向零取整大 1
    return a / b + (a % b != 0 && (a ^ b) >= 0);
}

void solve()
{
    ll a, b;
    cin >> a >> b;
    cout << floorDiv(a, b) << ' ' << ceilDiv(a, b) << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

处理负数范围的除法、二分取中点、数论分块。只做几次整数运算，时间复杂度 O(1)。
