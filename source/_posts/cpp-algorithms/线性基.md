线性基把每个数看成二进制向量，异或就是模 2 加法。插入时从高位往低位扫，如果这一位已经有主元，就异或掉；没有就把 x 放在这一位。这样每个二进制位最多留一个数，任意子集异或值都能由基里的数异或出来。求最大异或时从高位贪心，异或后变大就异或。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int B = 60;
ll basis[B];

void initBasis()
{
    memset(basis, 0, sizeof(basis));
}

// 插入成功返回 true，已经能被表示返回 false
bool insertBasis(ll x)
{
    for (int b = B - 1; b >= 0; b--)
    {
        if (((x >> b) & 1) == 0) continue;
        if (basis[b] == 0) // 这一位还没有主元，x 可以放进来
        {
            basis[b] = x;
            return true;
        }
        x ^= basis[b]; // 消掉第 b 位，继续往下找
    }
    return false;
}

// 子集异或最大值
ll maxXor()
{
    ll res = 0;
    for (int b = B - 1; b >= 0; b--)
        // 高位贪心：异或后变大就取
        if ((res ^ basis[b]) > res) res ^= basis[b];
    return res;
}

// 判断 x 能否由子集异或得到
bool canRepresent(ll x)
{
    for (int b = B - 1; b >= 0; b--)
    {
        if (((x >> b) & 1) == 0) continue;
        if (basis[b] == 0) return false; // 这一位消不掉，表示不了
        x ^= basis[b];
    }
    return true;
}

void solve()
{
    int n;
    cin >> n;

    initBasis();
    for (int i = 0; i < n; i++)
    {
        ll x;
        cin >> x;
        insertBasis(x);
    }

    cout << maxXor() << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

求子集最大异或、判断一个数能否由子集异或得到、维护异或空间。每次插入或查询 O(60)，空间 O(60)。
