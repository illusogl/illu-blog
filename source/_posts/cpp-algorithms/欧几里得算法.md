gcd 用辗转相除，b 为 0 时 a 就是最大公约数。扩展欧几里得在递归返回时，用上一层的 x、y 推出这一层的解：x1 = y，y1 = x - y * (a / b)。这样能得到 ax + by = gcd(a,b) 的一组整数解。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

// 普通 gcd
ll gcd(ll a, ll b)
{
    // 辗转相除：gcd(a,b) = gcd(b,a%b)
    return b == 0 ? a : gcd(b, a % b);
}

// 返回 {g, x, y}，满足 a*x + b*y = g = gcd(a, b)
tuple<ll, ll, ll> exgcd(ll a, ll b)
{
    if (b == 0) return {a, 1, 0}; // ax + by = a 的一组解
    auto [g, x, y] = exgcd(b, a % b); // 先递归求下一层
    // 用下一层的解反推这一层
    return {g, y, x - (a / b) * y};
}

void solve()
{
    ll a, b;
    cin >> a >> b;
    auto [g, x, y] = exgcd(a, b);
    cout << g << ' ' << x << ' ' << y << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

求最大公约数、最小公倍数、解同余方程，也能求模逆元。递归层数是 O(log min(a,b))，空间也是 O(log min(a,b))。
