任取一点 r，离它最远的点一定是直径的一端 x；再从 x 找最远的点 y，xy 就是一条直径。树边权为 1 时，用两次 DFS 就能求出端点和长度。下面的代码先读一棵树，第一次 DFS 找 x，第二次 DFS 找 y，同时得到直径长度。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

vector<vector<int>> g;
vector<int> dist;

// 从 u 出发 DFS，p 是父亲，d 是当前深度
void dfs(int u, int p, int d)
{
    dist[u] = d;
    for (int v : g[u])
        if (v != p) dfs(v, u, d + 1);
}

void solve()
{
    int n;
    cin >> n;

    g.assign(n + 1, {});
    for (int i = 0; i < n - 1; i++)
    {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    dist.assign(n + 1, 0);
    dfs(1, 0, 0); // 第一次：找离 1 最远的点 x
    // max_element 返回迭代器，减去 begin 得到下标
    int x = (int)(max_element(dist.begin() + 1, dist.end()) - dist.begin());

    dfs(x, 0, 0); // 第二次：从 x 找最远的点 y，dist[y] 就是直径
    int y = (int)(max_element(dist.begin() + 1, dist.end()) - dist.begin());

    cout << x << ' ' << y << ' ' << dist[y] << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

求树的直径、直径端点和树上最远点对。两次 DFS 各遍历整棵树，时间复杂度 O(n)，空间 O(n)；递归深度也是 O(n)，链很长时改用 BFS 或加大栈。
