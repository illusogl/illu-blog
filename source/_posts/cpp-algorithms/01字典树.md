把每个数按二进制从高位到低位插入 Trie，每个节点存两个儿子，分别对应这一位是 0 还是 1。查询最大异或时，尽量走与当前位相反的分支；查询与 x 异或后不小于 k 的个数时，整棵满足条件的子树可以直接累加计数。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int B = 30;

struct Node
{
    int ch[2] = {0, 0};
    int cnt = 0;
};

vector<Node> tr;

// 插入一个数
void insert(ll x)
{
    int u = 0;
    for (int b = B - 1; b >= 0; b--)
    {
        int c = (x >> b) & 1;
        if (!tr[u].ch[c]) // 没有这个儿子就新建
        {
            tr[u].ch[c] = (int)tr.size();
            tr.emplace_back();
        }
        u = tr[u].ch[c];
        tr[u].cnt++; // 记录经过这个节点的数的个数
    }
}

// 已插入的数中，与 x 异或的最大值
ll maxXor(ll x)
{
    int u = 0;
    ll res = 0;
    for (int b = B - 1; b >= 0; b--)
    {
        int c = (x >> b) & 1;
        int v = tr[u].ch[c ^ 1]; // 优先走相反位，异或结果更大
        if (v)
        {
            res |= 1LL << b; // 这一位异或结果为 1
            u = v;
        }
        else
        {
            u = tr[u].ch[c];
        }
    }
    return res;
}

// 已插入的数中，有多少个 y 满足 (x ^ y) >= k
ll countAtLeast(ll x, ll k)
{
    int u = 0;
    ll cur = 0, res = 0;
    for (int b = B - 1; b >= 0; b--)
    {
        int c = (x >> b) & 1;
        int v = tr[u].ch[c ^ 1];
        if (v)
        {
            // 整棵子树都满足 (x ^ y) >= k
            if (cur + (1LL << b) >= k) res += tr[v].cnt;
            // 这一位取 1 仍不够，继续深入
            else
            {
                cur += 1LL << b;
                u = v;
                continue;
            }
        }
        u = tr[u].ch[c];
        if (!u) break;
    }
    return res;
}

void solve()
{
    int n;
    cin >> n;

    tr.assign(1, Node());
    for (int i = 0; i < n; i++)
    {
        ll x;
        cin >> x;
        insert(x);
    }

    ll x, k;
    cin >> x >> k;
    cout << maxXor(x) << ' ' << countAtLeast(x, k) << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

求最大异或对、统计异或值不小于 k 的个数；异或第 k 大可以在 Trie 上按位统计子树大小来求。每个数插入和查询都是 O(位数)，这里取 30 位；空间 O(30n)。
