把每个数按二进制从高位到低位插入 Trie，每个节点存两个儿子，分别对应这一位是 0 还是 1。查询与 x 异或后不小于 k 的个数时，沿着 x 的位走：如果另一侧的子树整棵都满足条件，就把它的计数直接加上；否则继续往能凑出更高位的地方走。

<!-- more -->

```cpp
int dic[31000005];
int di[31000005][2];
int T = 30;
ll tot = 0;
void insert(ll x)
{
    ll u = 0;
    for(int i = T - 1; i >= 0; i--)
    {
        int b = (x >> i) & 1;
        if (!di[u][b]) di[u][b] = ++tot;
        u = di[u][b];
        dic[u]++;
    }
}
ll query(ll x, ll k)
{
    int u = 0;
    ll cur = 0;
    ll res = 0;
    for (int i = T - 1; i >= 0; i--)
    {
        int b = (x >> i) & 1;
        int o = di[u][b ^ 1];
        if (o)
        {
            if (cur + (1LL << i) >= k) res += dic[o];      // 整个子树都 >= k
            else { cur += 1LL << i; u = o; continue; }     // 走 b 的子树全 < k, 转进 o
        }
        u = di[u][b];
        if (!u) break;
    }
    return res;
}
```

求最大异或对、异或第 k 大、统计异或值不小于 k 的个数。每个数插入和查询都是 O(位数)，这里取 30 位；空间 O(30n)。
