n 个点 n 条边的连通图有且只有一个环。把所有度数为 1 的点放进队列，依次删掉它们，并让邻居的度数减一；邻居变成叶子后也入队。删到最后还留在图里的点，就是环上的点。

<!-- more -->

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

void solve()
{
    int n;
    cin >> n;

    vector<vector<int>> g(n + 1);
    vector<int> deg(n + 1, 0);
    for (int i = 0; i < n; i++)
    {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
        deg[u]++;
        deg[v]++;
    }

    // 拓扑排序剥叶子，剩下的就是环
    vector<bool> onCycle(n + 1, true);
    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (deg[i] == 1) q.push(i);

    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        onCycle[u] = false; // u 被剥掉，不在环上
        for (int v : g[u])
        {
            if (!onCycle[v]) continue;
            deg[v]--; // 相当于删掉 u
            if (deg[v] == 1) q.push(v); // 邻居变成叶子，继续剥
        }
    }

    for (int i = 1; i <= n; i++)
        if (onCycle[i]) cout << i << ' ';
    cout << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

用来找基环树的环，之后可以在环上做 DP 或断环成树。每个点、每条边只处理一次，时间复杂度 O(n)，空间复杂度 O(n)。
