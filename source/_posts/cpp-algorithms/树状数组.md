树状数组用 i & -i 取出 lowbit，把下标拆成若干个二进制区间。修改 a[i] 时，从 i 开始不断加上 lowbit，沿途的节点都要更新；求前缀和时，从 i 开始不断减去 lowbit，把经过的节点累加。build 利用已经算好的节点，可以在线性时间里建好整棵树。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Fenwick
{
    int n;
    vector<ll> bit;

    Fenwick(int n = 0)
    {
        init(n);
    }

    void init(int n_)
    {
        n = n_;
        bit.assign(n + 1, 0);
    }

    // a[i] += v：i 的父节点是 i + lowbit(i)
    void add(int i, ll v)
    {
        for (; i <= n; i += i & -i) bit[i] += v; // 向上更新
    }

    // 前缀和：每次去掉最低位的 1
    ll sum(int i) const
    {
        ll res = 0;
        for (; i > 0; i -= i & -i) res += bit[i]; // 向下累加
        return res;
    }

    // a[l] + ... + a[r]
    ll rangeSum(int l, int r) const
    {
        if (l > r) return 0;
        // 区间和 = 两个前缀和相减
        return sum(r) - sum(l - 1);
    }

    // O(n) 建树，a 下标从 1 开始
    void build(const vector<ll>& a)
    {
        init((int)a.size() - 1);
        for (int i = 1; i <= n; i++)
        {
            bit[i] += a[i];
            // 把当前节点的值累加给父亲
            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];
        }
    }
};

// 逆序对：从右往左扫，inv += fw.sum(p[i] - 1); fw.add(p[i], 1);

void solve()
{
    int n, q;
    cin >> n >> q;

    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    Fenwick fw;
    fw.build(a);

    while (q--)
    {
        int op;
        cin >> op;
        if (op == 1)
        {
            int i;
            ll x;
            cin >> i >> x;
            fw.add(i, x);
        }
        else
        {
            int l, r;
            cin >> l >> r;
            cout << fw.rangeSum(l, r) << '\n';
        }
    }
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

支持单点修改、前缀和、区间和，也能求逆序对。每次修改或查询 O(log n)，建树 O(n)，空间 O(n)。
