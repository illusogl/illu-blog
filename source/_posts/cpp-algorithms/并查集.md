并查集用 parent 数组记录每个点的父亲。Find 一路往上找根，顺便把路径上的点直接挂到根上；Union 先找两个根，再按大小合并，把小树挂到大树下。unite 返回 false 说明两个点已经连通，再加边就会形成环。

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct DSU
{
    vector<int> p, sz;

    DSU(int n = 0)
    {
        init(n);
    }

    void init(int n)
    {
        p.resize(n + 1);
        sz.assign(n + 1, 1);
        iota(p.begin(), p.end(), 0); // 一开始每个点的父亲是自己
    }

    int find(int x)
    {
        // 路径压缩：把沿途的点直接挂到根上
        return p[x] == x ? x : p[x] = find(p[x]);
    }

    // 合并成功返回 true，已经在同一集合返回 false
    bool unite(int a, int b)
    {
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b); // 小树挂到大树下
        p[b] = a;
        sz[a] += sz[b];
        return true;
    }
};

void solve()
{
    int n, m;
    cin >> n >> m;

    DSU dsu(n);
    int components = n;
    for (int i = 0; i < m; i++)
    {
        int u, v;
        cin >> u >> v;
        if (dsu.unite(u, v)) components--; // 合并成功，连通块少一个
        // else: u 和 v 已经连通，这条边会成环
    }

    cout << components << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();
    return 0;
}
```

判断连通性、检测无向图里的环、统计连通块，也用于 Kruskal 求最小生成树。路径压缩加按大小合并后，单次操作接近 O(1)，空间 O(n)。
